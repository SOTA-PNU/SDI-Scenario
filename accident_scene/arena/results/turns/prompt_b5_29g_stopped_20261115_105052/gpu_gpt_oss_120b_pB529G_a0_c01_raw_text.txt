import math
import heapq

def plan_path(start_xy, goal_xy, clearance_m):
    """
    A* on the occupancy grid (8‑connectivity) with segment_free edge checks.
    Returns a list of world (x, y) points including the start and goal
    (or their connectors) that passes validate_plan, otherwise None.
    """
    # ---- grid generation -------------------------------------------------
    plan_stage("grid")
    blocked, origin, res = occupancy_grid(clearance_m, 0.5)

    # ---- start / goal cells ---------------------------------------------
    try:
        start_cell = cell_of(start_xy, origin, res)
        goal_cell = cell_of(goal_xy, origin, res)
    except Exception:
        return None

    ny, nx = blocked.shape
    def in_bounds(c):
        ix, iy = c
        return 0 <= ix < nx and 0 <= iy < ny

    # start or goal inside a blocked cell -> cannot plan at this clearance
    if not in_bounds(start_cell) or not in_bounds(goal_cell):
        return None
    if blocked[start_cell[1], start_cell[0]] or blocked[goal_cell[1], goal_cell[0]]:
        return None

    # ---- A* search -------------------------------------------------------
    plan_stage("search")
    start_pt = xy_of(start_cell, origin, res)
    goal_pt = xy_of(goal_cell, origin, res)

    # heuristic: Euclidean distance in world space
    def heuristic(cell):
        wx, wy = xy_of(cell, origin, res)
        return math.hypot(wx - goal_pt[0], wy - goal_pt[1])

    frontier = [(heuristic(start_cell), 0.0, start_cell)]
    came_from = {start_cell: None}
    cost_so_far = {start_cell: 0.0}
    # cache segment_free results to avoid duplicate geometry checks
    seg_cache = {}

    while frontier:
        _, g, current = heapq.heappop(frontier)

        if current == goal_cell:
            break

        cx, cy = current
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                neighbor = (cx + dx, cy + dy)
                if not in_bounds(neighbor):
                    continue
                if blocked[neighbor[1], neighbor[0]]:
                    continue

                # world coordinates of the two cells
                p = xy_of(current, origin, res)
                q = xy_of(neighbor, origin, res)

                # edge clearance test (cached)
                key = (p, q)
                free = seg_cache.get(key)
                if free is None:
                    free = segment_free(p, q, clearance_m)
                    seg_cache[key] = free
                if not free:
                    continue

                # movement cost (real Euclidean distance)
                step_cost = math.hypot(q[0] - p[0], q[1] - p[1])
                new_cost = g + step_cost
                if neighbor not in cost_so_far or new_cost < cost_so_far[neighbor]:
                    cost_so_far[neighbor] = new_cost
                    priority = new_cost + heuristic(neighbor)
                    heapq.heappush(frontier, (priority, new_cost, neighbor))
                    came_from[neighbor] = current

    # ---- reconstruct path ------------------------------------------------
    if goal_cell not in came_from:
        return None  # no path found

    raw_path = []
    cur = goal_cell
    while cur is not None:
        raw_path.append(xy_of(cur, origin, res))
        cur = came_from[cur]
    raw_path.reverse()  # from start to goal

    # ---- (optional) simplification ---------------------------------------
    # For determinism we skip aggressive simplification; we just pass the raw path.
    plan_stage("simplify", input_points=len(raw_path), output_points=len(raw_path))
    simplified_path = raw_path

    # ---- validation -------------------------------------------------------
    plan_stage("validate", input_points=len(simplified_path))
    result = validate_plan(simplified_path)
    if result.get("accepted"):
        return result["path"]  # already a list/tuple of (x, y)
    return None


def lookahead_target(path, pose):
    """
    Return a point on the given polyline that lies 0.3‑2.0 m ahead of the robot
    projection onto the path. If insufficient forward distance exists, return
    None so the base will request a checked‑zero.
    """
    # helper: compute cumulative arc‑lengths of the path
    cum_len = [0.0]
    for a, b in zip(path, path[1:]):
        seg = math.hypot(b[0] - a[0], b[1] - a[1])
        cum_len.append(cum_len[-1] + seg)

    # find nearest point on the polyline (projection)
    rx, ry = pose[0], pose[1]
    best_dist = float("inf")
    best_s = 0.0
    for i, (a, b) in enumerate(zip(path, path[1:])):
        ax, ay = a
        bx, by = b
        dx, dy = bx - ax, by - ay
        seg_len2 = dx * dx + dy * dy
        if seg_len2 == 0.0:
            continue
        t = ((rx - ax) * dx + (ry - ay) * dy) / seg_len2
        t = max(0.0, min(1.0, t))
        proj_x = ax + t * dx
        proj_y = ay + t * dy
        d = math.hypot(rx - proj_x, ry - proj_y)
        s = cum_len[i] + t * math.sqrt(seg_len2)
        if d < best_dist - 1e-12 or (abs(d - best_dist) < 1e-12 and s < best_s):
            best_dist = d
            best_s = s

    # forward distance limits
    MIN_AHEAD = _BASE_LOOP_CFG["target_min_ahead_m"]
    MAX_AHEAD = _BASE_LOOP_CFG["target_max_ahead_m"]

    remaining = cum_len[-1] - best_s
    if remaining < MIN_AHEAD - 1e-12:
        return None  # not enough path ahead

    ahead = max(MIN_AHEAD, min(MAX_AHEAD, remaining))
    target_s = best_s + ahead
    if target_s > cum_len[-1]:
        target_s = cum_len[-1]

    # locate the segment containing target_s
    for i in range(len(cum_len) - 1):
        if cum_len[i] <= target_s <= cum_len[i + 1]:
            a = path[i]
            b = path[i + 1]
            seg_len = cum_len[i + 1] - cum_len[i]
            if seg_len == 0.0:
                return tuple(a)
            t = (target_s - cum_len[i]) / seg_len
            return (a[0] + t * (b[0] - a[0]), a[1] + t * (b[1] - a[1]))
    # fallback (should not happen)
    return tuple(path[-1])


def follow_cmd(pose, target_xy):
    """
    Pure pursuit law:
        v = 0                     if |e| > 0.35 rad
        v = min(1.5, 2.5·dist+0.5) otherwise
        w = clamp(7.5·e, -2.0, 2.0)
    where e is the wrapped yaw error to the target.
    """
    tx, ty = target_xy
    x, y, yaw = pose
    dx, dy = tx - x, ty - y
    dist = math.hypot(dx, dy)
    angle_to_target = math.atan2(dy, dx)
    e = angle_to_target - yaw
    e = math.atan2(math.sin(e), math.cos(e))  # wrap to [-π, π]

    if abs(e) > 0.35:
        v = 0.0
    else:
        v = min(1.5, 2.5 * dist + 0.5)

    w = max(-2.0, min(2.0, 7.5 * e))
    return v, w
