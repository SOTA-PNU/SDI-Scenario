import math

def plan_path(start_xy, goal_xy, clearance_m):
    """Simple deterministic planner.
    Tries a straight line, then an L‑shaped path (vertical‑then‑horizontal
    and horizontal‑then‑vertical), and finally a two‑segment path via the
    midpoint of the start‑goal line.  The first candidate whose segments all
    satisfy ``segment_free`` at the requested clearance is returned (including
    the exact start and goal points).  The candidate path is validated with
    ``validate_plan``; if validation fails ``None`` is returned so the base will
    retry with a smaller clearance.
    """
    # generate grid only to satisfy the API contract (not used directly)
    plan_stage("grid")
    occupancy_grid(clearance_m, 0.5)

    def try_path(pts):
        path = [tuple(start_xy)] + [tuple(p) for p in pts] + [tuple(goal_xy)]
        plan_stage("simplify", input_points=len(path), output_points=len(path))
        plan_stage("validate", input_points=len(path))
        result = validate_plan(path)
        if result.get("accepted"):
            return result["path"]
        return None

    # 1) direct
    if segment_free(start_xy, goal_xy, clearance_m):
        direct = try_path([])
        if direct:
            return direct

    sx, sy = start_xy
    gx, gy = goal_xy

    # 2) vertical then horizontal
    mid_vh = (sx, gy)
    if segment_free(start_xy, mid_vh, clearance_m) and segment_free(mid_vh, goal_xy, clearance_m):
        vh = try_path([mid_vh])
        if vh:
            return vh

    # 3) horizontal then vertical
    mid_hv = (gx, sy)
    if segment_free(start_xy, mid_hv, clearance_m) and segment_free(mid_hv, goal_xy, clearance_m):
        hv = try_path([mid_hv])
        if hv:
            return hv

    # 4) via midpoint of start‑goal line (two segments)
    mx, my = (sx + gx) * 0.5, (sy + gy) * 0.5
    # try vertical then horizontal via midpoint
    mid1 = (sx, my)
    mid2 = (gx, my)
    if (segment_free(start_xy, mid1, clearance_m) and
        segment_free(mid1, mid2, clearance_m) and
        segment_free(mid2, goal_xy, clearance_m)):
        via_mid = try_path([mid1, mid2])
        if via_mid:
            return via_mid

    # 5) try a few additional fixed waypoints (north‑west, north‑east, etc.)
    # These are heuristic guesses that avoid the dense central box region.
    candidates = [
        (sx, sy + 5.0, gx, sy + 5.0),   # go north 5 m, then east
        (sx, sy + 10.0, gx, sy + 10.0), # go north 10 m, then east
        (sx - 5.0, sy, sx - 5.0, gy),   # go west then north
        (sx - 5.0, sy, gx, sy),         # go west then east
        (sx, sy - 5.0, gx, sy - 5.0),   # go south then east
    ]
    for cx1, cy1, cx2, cy2 in candidates:
        p1 = (cx1, cy1)
        p2 = (cx2, cy2)
        if (segment_free(start_xy, p1, clearance_m) and
            segment_free(p1, p2, clearance_m) and
            segment_free(p2, goal_xy, clearance_m)):
            cand = try_path([p1, p2])
            if cand:
                return cand

    # No candidate succeeded – let the base retry with a lower clearance.
    return None


def lookahead_target(path, pose):
    """Return a point 0.3 – 2.0 m ahead of the robot projection onto *path*.
    If insufficient forward distance remains, ``None`` is returned so the base
    will request a checked‑zero.
    """
    # cumulative arc‑lengths of the polyline
    cum_len = [0.0]
    for a, b in zip(path, path[1:]):
        seg = math.hypot(b[0] - a[0], b[1] - a[1])
        cum_len.append(cum_len[-1] + seg)

    rx, ry = pose[0], pose[1]
    best_dist = float("inf")
    best_s = 0.0
    for i, (a, b) in enumerate(zip(path, path[1:])):
        ax, ay = a
        bx, by = b
        dx, dy = bx - ax, by - ay
        seg_len2 = dx * dx + dy * dy
        if seg_len2 == 0.0:
            continue
        t = ((rx - ax) * dx + (ry - ay) * dy) / seg_len2
        t = max(0.0, min(1.0, t))
        proj_x = ax + t * dx
        proj_y = ay + t * dy
        d = math.hypot(rx - proj_x, ry - proj_y)
        s = cum_len[i] + t * math.sqrt(seg_len2)
        if d < best_dist - 1e-12 or (abs(d - best_dist) < 1e-12 and s < best_s):
            best_dist = d
            best_s = s

    MIN_AHEAD = _BASE_LOOP_CFG["target_min_ahead_m"]
    MAX_AHEAD = _BASE_LOOP_CFG["target_max_ahead_m"]
    remaining = cum_len[-1] - best_s
    if remaining < MIN_AHEAD - 1e-12:
        return None
    ahead = max(MIN_AHEAD, min(MAX_AHEAD, remaining))
    target_s = best_s + ahead
    if target_s > cum_len[-1]:
        target_s = cum_len[-1]

    for i in range(len(cum_len) - 1):
        if cum_len[i] <= target_s <= cum_len[i + 1]:
            a = path[i]
            b = path[i + 1]
            seg_len = cum_len[i + 1] - cum_len[i]
            if seg_len == 0.0:
                return tuple(a)
            t = (target_s - cum_len[i]) / seg_len
            return (a[0] + t * (b[0] - a[0]), a[1] + t * (b[1] - a[1]))
    return tuple(path[-1])


def follow_cmd(pose, target_xy):
    """Pure‑pursuit law as specified."""
    tx, ty = target_xy
    x, y, yaw = pose
    dx, dy = tx - x, ty - y
    dist = math.hypot(dx, dy)
    ang = math.atan2(dy, dx)
    e = math.atan2(math.sin(ang - yaw), math.cos(ang - yaw))
    if abs(e) > 0.35:
        v = 0.0
    else:
        v = min(1.5, 2.5 * dist + 0.5)
    w = max(-2.0, min(2.0, 7.5 * e))
    return v, w
